cmake_minimum_required(VERSION 3.20)
project(CrankJam VERSION 0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(CJ_BUILD_HOST "Build SDL2 desktop transport" ON)
option(CJ_BUILD_HEADLESS "Build the SDK-free command-line transport" ON)
option(CJ_BUILD_TESTS "Build portable runtime tests" ON)
option(CJ_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(CJ_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(LUA_DIR ${CMAKE_CURRENT_SOURCE_DIR}/vendor/lua-5.4.9/src)
set(LUA_FILES lapi.c lcode.c lctype.c ldebug.c ldo.c ldump.c lfunc.c lgc.c llex.c lmem.c
  lobject.c lopcodes.c lparser.c lstate.c lstring.c ltable.c ltm.c lundump.c lvm.c lzio.c
  lauxlib.c lbaselib.c lmathlib.c lstrlib.c ltablib.c lutf8lib.c)
list(TRANSFORM LUA_FILES PREPEND "${LUA_DIR}/")
add_library(cj_lua STATIC ${LUA_FILES})
target_include_directories(cj_lua PUBLIC ${LUA_DIR})
add_library(crankjam STATIC src/runtime.c src/graphics.c src/sound.c)
target_include_directories(crankjam PUBLIC include PRIVATE src)
target_link_libraries(crankjam PRIVATE cj_lua)
if(NOT MSVC)
  target_compile_options(crankjam PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  target_link_libraries(crankjam PUBLIC m)
endif()
if(CJ_BUILD_HEADLESS)
  add_executable(crankjam-headless host/main.c)
  target_compile_definitions(crankjam-headless PRIVATE CJ_NO_SDL=1)
  target_compile_options(crankjam-headless PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  target_link_libraries(crankjam-headless PRIVATE crankjam)
endif()
if(CJ_BUILD_HOST)
  find_program(SDL2_CONFIG sdl2-config REQUIRED HINTS /opt/homebrew/bin)
  execute_process(COMMAND ${SDL2_CONFIG} --cflags OUTPUT_VARIABLE SDL_CFLAGS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  execute_process(COMMAND ${SDL2_CONFIG} --libs OUTPUT_VARIABLE SDL_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  separate_arguments(SDL_CFLAGS UNIX_COMMAND "${SDL_CFLAGS}")
  separate_arguments(SDL_LIBS UNIX_COMMAND "${SDL_LIBS}")
  add_executable(crankjam-host host/main.c)
  target_compile_options(crankjam-host PRIVATE ${SDL_CFLAGS} -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  target_link_libraries(crankjam-host PRIVATE crankjam ${SDL_LIBS})
endif()
if(CJ_BUILD_TESTS)
  enable_testing()
  add_executable(crankjam-tests tests/runtime/test_runtime.c)
  target_include_directories(crankjam-tests PRIVATE src)
  target_compile_options(crankjam-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  target_link_libraries(crankjam-tests PRIVATE crankjam cj_lua)
  add_test(NAME runtime COMMAND crankjam-tests)
  set_tests_properties(runtime PROPERTIES TIMEOUT 20)
endif()
