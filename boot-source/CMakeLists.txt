cmake_minimum_required(VERSION 3.20)
project(CrankfruitBoot LANGUAGES C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT DEFINED CRANKJAM_ENGINE_DIR)
  message(FATAL_ERROR "Set CRANKJAM_ENGINE_DIR to the portable engine source directory")
endif()
set(CJ_BUILD_HOST OFF CACHE BOOL "No SDL needed for boot rendering")
set(CJ_BUILD_HEADLESS OFF CACHE BOOL "Only the boot renderer is needed here")
set(CJ_BUILD_TESTS ON CACHE BOOL "Keep core tests available")
add_subdirectory("${CRANKJAM_ENGINE_DIR}" engine)
add_executable(render-boot render_boot.c)
target_link_libraries(render-boot PRIVATE crankjam)
if(CJ_SANITIZE AND NOT MSVC)
  target_compile_options(render-boot PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(render-boot PRIVATE -fsanitize=address,undefined)
endif()
